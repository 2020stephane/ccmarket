/**
 * @fileoverview Déclaration des routes CRUD sur les annonces,
 * utilisées pour les tests via Postman.
 * @module postman
 * @project ccmarket
 * @version 1.0.0
 * @date 2026-06-24
 * @author Stephane Brisse
 * @license MIT
 * @requires express
 * @requires ../controllers/postmanControllers.js
 */

import express from 'express';
import {
   getAnnonces,
   getAnnonceById,
   postAnnonce,
   putAnnonce,
   patchAnnonce,
   deleteAnnonce,
} from '../controllers/postmanControllers.js';

/**
 * Routeur Express dédié au CRUD de test Postman.
 * @type {express.Router}
 * @const
 */
const router = express.Router();

/**
 * Route : récupère la liste de toutes les annonces.
 * @name GET/
 * @function
 * @param {string} path - `/`
 * @param {Function} getAnnonces - Contrôleur renvoyant l'ensemble des annonces.
 */
router.get('/', getAnnonces);

/**
 * Route : récupère une annonce à partir de son identifiant.
 * @name GET/:id
 * @function
 * @param {string} path - `/:id`
 * @param {Function} getAnnonceById - Contrôleur renvoyant l'annonce ciblée par `id`.
 */
router.get('/:id', getAnnonceById);

/**
 * Route : crée une nouvelle annonce.
 * @name POST/
 * @function
 * @param {string} path - `/`
 * @param {Function} postAnnonce - Contrôleur créant l'annonce en base.
 */
router.post('/', postAnnonce);

/**
 * Route : remplace intégralement une annonce existante.
 * @name PUT/:id
 * @function
 * @param {string} path - `/:id`
 * @param {Function} putAnnonce - Contrôleur remplaçant l'annonce ciblée par `id`.
 */
router.put('/:id', putAnnonce);

/**
 * Route : modifie partiellement une annonce existante.
 * @name PATCH/:id
 * @function
 * @param {string} path - `/:id`
 * @param {Function} patchAnnonce - Contrôleur mettant à jour l'annonce ciblée par `id`.
 */
router.patch('/:id', patchAnnonce);

/**
 * Route : supprime une annonce existante.
 * @name DELETE/:id
 * @function
 * @param {string} path - `/:id`
 * @param {Function} deleteAnnonce - Contrôleur supprimant l'annonce ciblée par `id`.
 */
router.delete('/:id', deleteAnnonce);

export default router;
